1 · Cómo arranca todo
Arranque

server.js

Acá lo que hacemos es poner en marcha la aplicación: cargamos el .env, conectamos a Mongo y, recién ahí, levantamos el servidor.

import 'dotenv/config';
import app from './app.js';
import { connectMongoDB } from './database/mongodb.js';

const PORT = process.env.PORT || 3000;

try {
  await connectMongoDB();      // 1. primero la base
  app.listen(PORT);            // 2. recién ahí, el servidor
} catch (error) {
  process.exit(1);             // si la base falla, no arranca
}
Idea clave: conectamos antes de levantar el servidor. Sin base, la app no arranca.
Base de datos

database/mongodb.js

Acá lo que hacemos es conectarnos a MongoDB con Mongoose, usando la URL del .env. Es la única parte que sabe cómo conectarse.

import mongoose from 'mongoose';
import dns from 'node:dns';

dns.setServers(['8.8.8.8', '1.1.1.1']);   // DNS de Google (fix de conexión)

export async function connectMongoDB() {
  await mongoose.connect(process.env.MONGO_URI);  // lee la URL del .env
}
Idea clave: la URL vive en el .env, fuera del código, para no exponer la contraseña.
Configuración

app.js

Acá lo que hacemos es configurar Express: los middlewares (leer JSON, CORS), montamos las rutas, y el manejo de errores. Es la "puerta de entrada" de toda request.

const app = express();

app.use(cors());               // permite el frontend (otro origen)
app.use(express.json());       // para leer el body JSON

app.use('/auth', authRouter);  // las requests entran por acá
app.use('/songs', songRouter);

app.use((err, req, res, next) => {          // manejo de errores central
  res.status(err.status || 500).json({ error: err.message });
});
Idea clave: toda request pasa primero por acá (middlewares), y de acá se reparte a la ruta que corresponda.
2 · Seguí una request al catálogo
POST /songs  →  song.routes → song.controller → song.service → (iTunes) → Song → MongoDB
Rutaspaso 1 · entra

routes/song.routes.js

La request entra por acá. Acá lo que hacemos es mapear la URL a la función del controller que la atiende.

router.get('/search', songController.buscar);   // GET  /songs/search?q=
router.get('/', songController.listar);         // GET  /songs
router.post('/', songController.importar);      // POST /songs
Controllerpaso 2 · traduce HTTP

controllers/song.controller.js

Acá lo que hacemos es traducir HTTP: leer el req, llamar al service, y responder con su código. Sin lógica de negocio.

export async function importar(req, res, next) {
  try {
    const { artista } = req.body;                 // lee el body
    if (!artista) throw httpError(400, 'Falta el campo "artista"');
    res.status(201).json(await songService.importarArtista(artista));  // delega y responde
  } catch (err) { next(err); }
}

export async function listar(req, res, next) {
  try { res.json(await songService.listarCanciones()); }
  catch (err) { next(err); }
}
Servicepaso 3 · la lógica

services/song.service.js

Acá vive la lógica del catálogo: importar sin duplicar, listar, buscar. No sabe qué es una request HTTP — solo trabaja con datos.

export async function importarArtista(artista) {
  const canciones = await buscarCancionesDeArtista(artista);   // pide a iTunes
  for (const c of canciones) {
    await Song.updateOne(
      { itunesId: c.itunesId },   // buscá una con este id
      { $setOnInsert: c },        // si NO existe, insertala
      { upsert: true }            // "upsert" = insertar-o-nada
    );
  }
}

export async function buscarEnCatalogo(q) {
  const regex = new RegExp(q, 'i');   // 'i' = ignora mayúsculas
  return Song.find({ $or: [{ title: regex }, { artist: regex }] }).limit(10);
}
Idea clave: el upsert por itunesId hace que reimportar no duplique.
Servicepaso 3b · habla con iTunes

services/itunes.service.js

El service del catálogo llama a este. Acá lo que hacemos es pedirle canciones a iTunes y traducirlas a la forma de nuestro Song.

const url = `https://itunes.apple.com/search?term=${artista}&entity=song&country=UY`;
const res = await fetch(url);
const data = await res.json();

return data.results
  .filter((r) => r.previewUrl)          // solo las que tienen audio
  .map((r) => ({                          // a NUESTROS nombres
    title: r.trackName,
    artist: r.artistName,
    previewUrl: r.previewUrl,
    year: new Date(r.releaseDate).getFullYear(),
    itunesId: String(r.trackId),
  }));
Idea clave: traducimos los nombres de iTunes a los nuestros. El resto del código no sabe que iTunes existió.
Modelopaso 4 · los datos

models/song.model.js

Acá lo que hacemos es definir la forma de una canción y darnos la herramienta (Song) para guardarla y buscarla en Mongo. El itunesId es único.

const songSchema = new mongoose.Schema({
  title:      { type: String, required: true },
  artist:     { type: String, required: true },
  previewUrl: { type: String, required: true },   // audio de 30s
  year:       { type: Number },                   // para la pista de año
  itunesId:   { type: String, required: true, unique: true },
}, { timestamps: true });

export const Song = mongoose.model('Song', songSchema);
Idea clave: el Schema es el molde; mongoose.model() lo vuelve la herramienta para trabajar los datos.
3 · Seguí una request de login
POST /auth/login  →  auth.routes → auth.controller → auth.service (bcrypt + JWT) → User → MongoDB
Rutaspaso 1 · entra

routes/auth.routes.js

La request de auth entra por acá. En /me metemos el authenticate en el medio: es la ruta protegida.

router.post('/register', authController.register);        // público
router.post('/login', authController.login);              // público
router.get('/me', authenticate, authController.me);       // PROTEGIDO
Idea clave: el authenticate antes del controller = esa ruta exige token válido.
Middlewareel guardia (solo en /me)

middleware/authenticate.js

En las rutas protegidas, la request pasa primero por acá. Acá lo que hacemos es verificar el token y guardar quién es en req.user.

export function authenticate(req, res, next) {
  const token = req.headers.authorization?.split(' ')[1];   // "Bearer xxx"
  try {
    req.user = jwt.verify(token, process.env.JWT_SECRET);   // valida la firma
    next();                                                 // deja pasar al controller
  } catch {
    next(httpError(401, 'Token inválido'));                 // si no, 401
  }
}
Controllerpaso 2 · traduce HTTP

controllers/auth.controller.js

Acá lo que hacemos es traducir HTTP para la auth: register, login, y me (el perfil del usuario del token).

export async function register(req, res, next) {
  try { res.status(201).json(await authService.register(req.body)); }
  catch (err) { next(err); }
}

export async function login(req, res, next) {
  try { res.json(await authService.login(req.body)); }   // { token, user }
  catch (err) { next(err); }
}

export async function me(req, res, next) {
  try { res.json(await authService.getPerfil(req.user.id)); }  // req.user viene del middleware
  catch (err) { next(err); }
}
Servicepaso 3 · la lógica

services/auth.service.js

Acá vive la lógica de auth. Al registrar, hasheamos la contraseña con bcrypt. Al loguear, la comparamos y firmamos un token JWT.

export async function register({ username, email, password }) {
  const passwordHash = await bcrypt.hash(password, 10);   // la ciframos
  return User.create({ username, email, passwordHash });
}

export async function login({ email, password }) {
  const user = await User.findOne({ email });
  const ok = await bcrypt.compare(password, user.passwordHash);   // comparamos
  if (!ok) throw httpError(401, 'Credenciales inválidas');

  const token = jwt.sign({ id: user._id }, process.env.JWT_SECRET);  // firmamos
  return { token };
}
Idea clave: la contraseña se guarda hasheada (irreversible); el login devuelve un token, no la sesión.
Modelopaso 4 · los datos

models/user.model.js

Acá lo que hacemos es definir la forma de un jugador. Guardamos el passwordHash (nunca la contraseña real) y su bestScore para el ranking.

const userSchema = new mongoose.Schema({
  username:     { type: String, required: true, unique: true },
  email:        { type: String, required: true, unique: true },
  passwordHash: { type: String, required: true },  // hasheada
  bestScore:    { type: Number, default: 0 },
  gamesPlayed:  { type: Number, default: 0 },
}, { timestamps: true });

export const User = mongoose.model('User', userSchema);
4 · Ayudantes (los usan varias capas)
Utils

utils/httpError.js

Acá lo que hacemos es un ayudante para crear errores con su código HTTP. Así los services lanzan un 401 o 404, y el manejador de errores del app.js lo traduce.

export function httpError(status, message) {
  const err = new Error(message);
  err.status = status;    // le "pega" el status al error
  return err;
}
Script

database/seedCatalog.js

Acá lo que hacemos es poblar el catálogo de una: recorremos una lista de artistas uruguayos y los importamos. Se corre con npm run seed.

await connectMongoDB();
